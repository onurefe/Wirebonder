\textbf{IF SOMETHING GOES WRONG}\par\vspace{2pt}
\begin{tabular}{@{}L{40mm}L{94mm}@{}}
\toprule
\msg{LOW BONDING POWER}   & Run \keycap{TEST}; check the scan window brackets resonance \\
\msg{COIL CURR. UNSTABLE} & Check the lever moves freely; lower the force \\
\msg{Z POS. NOT SETTLED}  & Clear the obstruction; run tachometer calibration \\
\msg{TIMEOUT}             & Workpiece out of reach, or contact sensor disconnected \\
\msg{ERROR SYSTEM LOCKED} & Press \keycap{RESET} \\
\msg{BUSY! TRY LATER}     & Wait for the current operation, then press again \\
\bottomrule
\end{tabular}

\par\vspace{1em}
\emph{Nothing is saved until you press \keycap{SAVE}.}
